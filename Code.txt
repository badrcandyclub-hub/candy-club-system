function detectPlatformName(rawPlatform) {
  var plt = (rawPlatform || "").toString().toLowerCase();
  if (plt.includes('ظˆط§طھط³') || plt.includes('whats')) return 'ظˆط§طھط³ط§ط¨';

  if (plt.includes('ظپظٹط³') || plt.includes('face') || plt.includes('facebook')) return 'ظپظٹط³ط¨ظˆظƒ';

function buildMonthlyReport(ordersData, targetMonth) {
  var monthTopProductsMap = {};
  var monthPlatformsMap   = { 'ظˆط§طھط³ط§ط¨': 0, 'ط§ظ†ط³طھط¬ط±ط§ظ…': 0, 'ظپظٹط³ط¨ظˆظƒ': 0, 'طھظٹظƒ طھظˆظƒ': 0, 'ط£ط®ط±ظ‰': 0 };
  var monthSales          = 0;
  var completedMonthCount = 0;
  var monthOrderCount     = 0;
  var returnedCount       = 0;
  var moderatorStats      = {}; // â­گ ط­ط³ط§ط¨ ط§ظ„ظ…ظˆط¯ط±ظٹطھظˆط±
  var driverStats         = {}; // â­گ ط­ط³ط§ط¨ ط§ظ„ظ…ظ†ط§ط¯ظٹط¨

  for (var j = 1; j < ordersData.length; j++) {
    var rowDate = ordersData[j][1] instanceof Date
      ? Utilities.formatDate(ordersData[j][1], "Africa/Cairo", "yyyy-MM-dd")
      : (ordersData[j][1] ? ordersData[j][1].toString().replace(/'/g, "").trim() : "");

    var status     = ordersData[j][18];
    var finalTotal = parseFloat(ordersData[j][17]) || 0;
    var shipping   = parseFloat(ordersData[j][16]) || 0;
    var mod        = (ordersData[j][20] || "").toString().trim(); // ط¹ظ…ظˆط¯ 21 ظ‡ظˆ ط§ظ„ظ…ظˆط¯ط±ظٹطھظˆط±
    var driver     = (ordersData[j][11] || "").toString().trim(); // ط¹ظ…ظˆط¯ 12 ظ‡ظˆ ط§ظ„ظ…ظ†ط¯ظˆط¨

    if (status !== "ظ…ط±طھط¬ط¹" && mod !== "") {
      if (!moderatorStats[mod]) {
          moderatorStats[mod] = { monthSales: 0, monthOrderCount: 0, totalSales: 0, totalCount: 0 };
      }
      moderatorStats[mod].totalSales += finalTotal;
      moderatorStats[mod].totalCount++;
      
      if (rowDate.startsWith(targetMonth)) {
         moderatorStats[mod].monthSales += finalTotal;
         moderatorStats[mod].monthOrderCount++;
      }
    }

    // â­گ ط­ط³ط§ط¨ ط§ظ„ظ…ظ†ط§ط¯ظٹط¨
    if (driver !== "" && (status === "طھظ… ط§ظ„طھظˆطµظٹظ„" || status === "طھظ… ط§ظ„طھظˆطµظٹظ„ ظˆظ…ظڈط­ط§ط³ط¨")) {
      if (!driverStats[driver]) {
          driverStats[driver] = { monthProfit: 0, monthOrderCount: 0, totalProfit: 0, totalCount: 0 };
      }
      driverStats[driver].totalProfit += shipping;
      driverStats[driver].totalCount++;
      
      if (rowDate.startsWith(targetMonth)) {
         driverStats[driver].monthProfit += shipping;
         driverStats[driver].monthOrderCount++;
      }
    }

    // ظپظ‚ط· ط³ط¬ظ„ط§طھ ط§ظ„ط´ظ‡ط± ط§ظ„ظ…ط·ظ„ظˆط¨
    if (!rowDate.startsWith(targetMonth)) continue;

    monthOrderCount++;

    if (status === "ظ…ط±طھط¬ط¹") {
      returnedCount++;
      continue; // ط§ظ„ظ…ط±طھط¬ط¹ط§طھ ظ„ط§ طھط¯ط®ظ„ ظپظٹ ط§ظ„ظ…ط¨ظٹط¹ط§طھ
    }

    monthSales += finalTotal;

    if (status === "طھظ… ط§ظ„طھظˆطµظٹظ„" || status === "طھظ… ط§ظ„طھظˆطµظٹظ„ ظˆظ…ظڈط­ط§ط³ط¨") {
      completedMonthCount++;
    }

    // â­گ طھط¬ظ…ظٹط¹ ط§ظ„ظ…ظ†طھط¬ط§طھ
    if (ordersData[j][13]) {
      var parts = ordersData[j][13].toString().split('\n');
      parts.forEach(function(p) {
        var m = p.match(/^(.+?)\s*-\s*ط§ظ„ظƒظ…ظٹط©:\s*(\d+)/);
        if (m) {
          var pName = m[1].trim();
          var pQty  = parseInt(m[2]) || 1;
          monthTopProductsMap[pName] = (monthTopProductsMap[pName] || 0) + pQty;
        }
      });
    }

    // â­گ طھط¬ظ…ظٹط¹ ط§ظ„ظ…ظ†طµط§طھ - ط¹ظ…ظˆط¯ ط§ظ„ظ…ظ†طµط© [3] ط£ظˆ ط§ظ„ظ…ظˆط¯ط±ظٹطھظˆط± [20] fallback
    var rawPlt   = ordersData[j][3] || ordersData[j][20] || "";
    var pltName  = detectPlatformName(rawPlt);
    monthPlatformsMap[pltName]++;
  }

  // طھط±طھظٹط¨ ط§ظ„ظ…ظ†طھط¬ط§طھ طھظ†ط§ط²ظ„ظٹط§ظ‹ ظˆط£ط®ط° ط£ظپط¶ظ„ 10
  var sortedProducts = Object.keys(monthTopProductsMap).map(function(k) {
    return { name: k, qty: monthTopProductsMap[k] };
  }).sort(function(a, b) { return b.qty - a.qty; }).slice(0, 10);

  return {
    monthTopProducts: sortedProducts,
    monthPlatforms:   monthPlatformsMap,
    monthSales:       monthSales,
    monthOrderCount:  monthOrderCount,
    completedMonthCount: completedMonthCount,
    returnedCount:    returnedCount,
    moderatorStats:   moderatorStats,
    driverStats:      driverStats
  };
}

function buildDailyStats(ordersData, targetDate) {
  var todayOrders  = 0;
  var todaySales   = 0;
  var completedToday = 0;

  for (var j = 1; j < ordersData.length; j++) {
    var rowDate = ordersData[j][1] instanceof Date
      ? Utilities.formatDate(ordersData[j][1], "Africa/Cairo", "yyyy-MM-dd")
      : (ordersData[j][1] ? ordersData[j][1].toString().replace(/'/g, "").trim() : "");

    if (rowDate !== targetDate) continue;

    var status     = ordersData[j][18];
    var finalTotal = parseFloat(ordersData[j][17]) || 0;

    if (status !== "ظ…ط±طھط¬ط¹") {
      todayOrders++;
      todaySales += finalTotal;
    }
    if (status === "طھظ… ط§ظ„طھظˆطµظٹظ„" || status === "طھظ… ط§ظ„طھظˆطµظٹظ„ ظˆظ…ظڈط­ط§ط³ط¨") {
      completedToday++;
    }
  }

  return { todayOrders: todayOrders, todaySales: todaySales, completedToday: completedToday };
}

function updateCustomerDB(sheet, name, phone, gov, address, amount, date) {
  var data       = sheet.getDataRange().getValues();
  var cleanPhone = phone.toString().replace(/'/g, "").trim();
  for (var i = 1; i < data.length; i++) {
    if (data[i][1].toString().replace(/'/g, "").trim() === cleanPhone) {
      var newCount = (parseInt(data[i][4]) || 0) + 1;
      var newTotal = (parseFloat(data[i][5]) || 0) + parseFloat(amount);
      sheet.getRange(i+1, 5, 1, 3).setValues([[newCount, newTotal, date]]);
      return;
    }
  }
  sheet.appendRow([name, "'" + cleanPhone, gov, address, 1, amount, date]);
}

function subtractFromCustomerDB(sheet, phone, amount) {
  if (!sheet) return;
  var data       = sheet.getDataRange().getValues();
  var cleanPhone = phone.toString().replace(/'/g, "").trim();
  for (var i = 1; i < data.length; i++) {
    if (data[i][1].toString().replace(/'/g, "").trim() === cleanPhone) {
      var cCount = parseInt(data[i][4]) || 1;
      var cTotal = parseFloat(data[i][5]) || 0;
      sheet.getRange(i+1, 5, 1, 2).setValues([[Math.max(0, cCount - 1), Math.max(0, cTotal - amount)]]);
      return;
    }
  }
}

function calculateFinancials() {
  var ss          = SpreadsheetApp.getActiveSpreadsheet();
  var ordersSheet = ss.getSheetByName("ًں“‌ ط§ظ„ط·ظ„ط¨ط§طھ");
  var finSheet    = ss.getSheetByName("ًں“ٹ ط§ظ„ط­ط³ط§ط¨ط§طھ ظˆط§ظ„ظ…ط§ظ„ظٹط©");
  
  if (!ordersSheet || !finSheet) return;

  var data     = ordersSheet.getDataRange().getValues();
  var accounts = {};

  for (var i = 1; i < data.length; i++) {
    var status    = data[i][18];
    var driver    = data[i][11];
    var payMethod = data[i][10] ? data[i][10].toString() : "";
    var shippingCost = parseFloat(data[i][16]) || 0;
    var finalTotal   = parseFloat(data[i][17]) || 0;
    
    var remainingCash = parseFloat(data[i][22]);
    if (isNaN(remainingCash)) remainingCash = finalTotal; 

    var isCash    = payMethod.indexOf("ظƒط§ط´ ًں’µ") !== -1 || (payMethod.includes("ظƒط§ط´") && !payMethod.includes("ظپظˆط¯ط§ظپظˆظ†"));
    var isDigital = payMethod.includes("ط¥ظ†ط³طھط§") || payMethod.includes("ط§ظ†ط³طھط§ط¨ط§ظٹ") || payMethod.includes("ظپظˆط¯ط§ظپظˆظ†") || payMethod.includes("طھط­ظˆظٹظ„") || payMethod.includes("ظ…ط­ظپط¸ط©");

    if ((status === "طھظ… ط§ظ„طھظˆطµظٹظ„" || status === "ظپظٹ ط§ظ„ط´ط­ظ†") && driver) {
      if (!accounts[driver]) {
        accounts[driver] = { ordersCount: 0, inTransit: 0, cashCollected: 0, shippingFees: 0, netDue: 0 };
      }
      
      if (status === "ظپظٹ ط§ظ„ط´ط­ظ†") {
         if (isCash) { accounts[driver].inTransit += remainingCash; }
      } 
      else if (status === "طھظ… ط§ظ„طھظˆطµظٹظ„") {
          accounts[driver].ordersCount++;
          if (isCash) {
            accounts[driver].cashCollected += remainingCash;
            accounts[driver].shippingFees  += shippingCost;
            accounts[driver].netDue        += (remainingCash - shippingCost); 
          } else if (isDigital) {
            accounts[driver].shippingFees += shippingCost;
            accounts[driver].netDue       -= shippingCost; 
          }
      }
    }
  }

  var lastRow = finSheet.getLastRow();
  if (lastRow > 1) {
    finSheet.getRange(2, 1, lastRow - 1, 7).clearContent();
  }

  var row = 2;
  for (var drv in accounts) {
    var acc        = accounts[drv];
    var statusText = acc.netDue > 0 ? "ط¹ط§ظٹط²ظٹظ† ظ…ظ†ظ‡" : (acc.netDue < 0 ? "ظ„ظٹظ‡ ط¹ظ†ط¯ظ†ط§" : "ط®ط§ظ„طµ");
    
    finSheet.getRange(row, 1, 1, 7).setValues([[
      drv, acc.ordersCount, acc.inTransit, acc.cashCollected, acc.shippingFees, acc.netDue, statusText 
    ]]);
    row++;
  }
}

function analyzeVIPs() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var ordersSheet = ss.getSheetByName("ًں“‌ ط§ظ„ط·ظ„ط¨ط§طھ");
  if (!ordersSheet) return [];
  
  var data = ordersSheet.getDataRange().getValues();
  var customersMap = {};
  
  for (var i = 1; i < data.length; i++) {
    var status = data[i][18];
    if (status === "طھظ… ط§ظ„طھظˆطµظٹظ„" || status === "طھظ… ط§ظ„طھظˆطµظٹظ„ ظˆظ…ظڈط­ط§ط³ط¨") {
      var phone = data[i][7].toString().replace(/'/g, "").trim();
      var name = data[i][4];
      var total = parseFloat(data[i][17]) || 0;
      
      if (!customersMap[phone]) {
        customersMap[phone] = { name: name, phone: phone, orderCount: 0, totalPaid: 0 };
      }
      customersMap[phone].orderCount++;
      customersMap[phone].totalPaid += total;
    }
  }
  
  var vipList = Object.keys(customersMap).map(function(key) {
    return customersMap[key];
  }).sort(function(a, b) {
    return b.orderCount - a.orderCount;
  });
  
  return vipList;
}
