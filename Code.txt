// ⭐ إضافة دالة التفاعل اليدوي: عشان لو غيرت الحالة من الإكسيل بإيدك، اللون يتغير والمالية تتحدث فوراً
function onEdit(e) {
  if (!e || !e.range) return;
  var sheet = e.range.getSheet();
  if (sheet.getName() !== "📝 الطلبات") return;
  
  var row = e.range.getRow();
  var col = e.range.getColumn();
  
  // العمود 19 هو "حالة الطلب" (S)
  if (col === 19 && row > 1) {
    var status = e.value;
    var oldStatus = e.oldValue;
    var rowRange = sheet.getRange(row, 1, 1, sheet.getLastColumn());
    
    if (status === "مرتجع") {
        rowRange.setBackground("#FEE2E2"); // أحمر فاتح
        if (oldStatus !== "مرتجع") {
            var phoneToSub = sheet.getRange(row, 8).getValue().toString().replace(/'/g, "").trim();
            var amtToSub = parseFloat(sheet.getRange(row, 18).getValue()) || 0;
            subtractFromCustomerDB(SpreadsheetApp.getActiveSpreadsheet().getSheetByName("👥 قاعدة العملاء"), phoneToSub, amtToSub);
        }
    } else if (status === "تم التوصيل ومُحاسب") {
        rowRange.setBackground("#E8F5E9"); // أخضر فاتح مريح
    } else if (status === "تم التوصيل") {
        rowRange.setBackground("#D4EDDA"); // أخضر باهت (قيد التحصيل)
    } else if (status === "في الشحن" || status === "قيد التجهيز") {
        rowRange.setBackground("#FFFFFF"); // أبيض طبيعي
    }
    
    // تحديث المالية في الخلفية


  calculateFinancials();
  }
}

// ⭐ V15.1: دالة مساعدة للتعرف على المنصة من النص

function detectPlatformName(rawPlatform) {
  var plt = (rawPlatform || "").toString().toLowerCase();
  if (plt.includes('واتس') || plt.includes('whats')) return 'واتساب';
  if (plt.includes('انستا') || plt.includes('insta') || plt.includes('انستجرام') || plt.includes('إنستجرام')) return 'انستجرام';
  if (plt.includes('فيس') || plt.includes('face') || plt.includes('facebook')) return 'فيسبوك';
  if (plt.includes('تيك') || plt.includes('tiktok')) return 'تيك توك';
  return 'أخرى';
}

// ⭐ V15.1: دالة مستقلة لحساب تقرير شهر محدد
// targetMonth = "2026-06" مثلاً
// ============================================================
function buildMonthlyReport(ordersData, targetMonth) {
  var monthTopProductsMap = {};
  var monthPlatformsMap   = { 'واتساب': 0, 'انستجرام': 0, 'فيسبوك': 0, 'تيك توك': 0, 'أخرى': 0 };
  var monthZonesMap       = {}; // ⭐ تحليل مناطق الشحن
  var monthSales          = 0;
  var completedMonthCount = 0;
  var monthOrderCount     = 0;
  var returnedCount       = 0;
  var moderatorStats      = {}; // ⭐ حساب المودريتور
  var driverStats         = {}; // ⭐ حساب المناديب

  for (var j = 1; j < ordersData.length; j++) {
    var rowDate = ordersData[j][1] instanceof Date
      ? Utilities.formatDate(ordersData[j][1], "Africa/Cairo", "yyyy-MM-dd")
      : (ordersData[j][1] ? ordersData[j][1].toString().replace(/'/g, "").trim() : "");

    var status     = ordersData[j][18];
    var finalTotal = parseFloat(ordersData[j][17]) || 0;
    var shipping   = parseFloat(ordersData[j][16]) || 0;
    var mod        = (ordersData[j][20] || "").toString().trim(); // عمود 21 هو المودريتور
    var driver     = (ordersData[j][11] || "").toString().trim(); // عمود 12 هو المندوب

    if (status !== "مرتجع" && mod !== "") {
      if (!moderatorStats[mod]) {
          moderatorStats[mod] = { monthSales: 0, monthOrderCount: 0, totalSales: 0, totalCount: 0 };
      }
      moderatorStats[mod].totalSales += finalTotal;
      moderatorStats[mod].totalCount++;
      
      if (rowDate.startsWith(targetMonth)) {
         moderatorStats[mod].monthSales += finalTotal;
         moderatorStats[mod].monthOrderCount++;
      }
    }

    // ⭐ حساب المناديب
    if (driver !== "" && (status === "تم التوصيل" || status === "تم التوصيل ومُحاسب")) {
      if (!driverStats[driver]) {
          driverStats[driver] = { monthProfit: 0, monthOrderCount: 0, totalProfit: 0, totalCount: 0 };
      }
      driverStats[driver].totalProfit += shipping;
      driverStats[driver].totalCount++;
      
      if (rowDate.startsWith(targetMonth)) {
         driverStats[driver].monthProfit += shipping;
         driverStats[driver].monthOrderCount++;
      }
    }

    // فقط سجلات الشهر المطلوب
    if (!rowDate.startsWith(targetMonth)) continue;

    monthOrderCount++;

    if (status === "مرتجع") {
      returnedCount++;
      continue; // المرتجعات لا تدخل في المبيعات
    }

    monthSales += finalTotal;

    if (status === "تم التوصيل" || status === "تم التوصيل ومُحاسب") {
      completedMonthCount++;
    }

    // ⭐ تحليل مناطق الشحن للشهر
    var zone = (ordersData[j][5] || "").toString().trim(); // عمود 6 (المنطقة/المحافظة)
    var orderType = (ordersData[j][9] || "").toString().trim();
    
    if (orderType === "استلام من الفرع" || orderType === "استلام من الفرع ") {
        zone = "استلام من الفرع";
    } else {
        if (!zone && ordersData[j][6]) {
           zone = ordersData[j][6].toString().split(/[-،,\n]/)[0].trim();
        }
        if (!zone) zone = "غير محددة";
    }

    if (!monthZonesMap[zone]) {
       monthZonesMap[zone] = { count: 0, totalShipping: 0 };
    }
    monthZonesMap[zone].count++;
    if (status !== "مرتجع") { // إضافة مصاريف الشحن فقط لو الأوردر متلغاش
        monthZonesMap[zone].totalShipping += shipping;
    }

    // ⭐ تجميع المنتجات
    if (ordersData[j][13]) {
      var parts = ordersData[j][13].toString().split('\n');
      parts.forEach(function(p) {
        var m = p.match(/^(.+?)\s*-\s*الكمية:\s*(\d+)/);
        if (m) {
          var pName = m[1].trim();
          var pQty  = parseInt(m[2]) || 1;
          monthTopProductsMap[pName] = (monthTopProductsMap[pName] || 0) + pQty;
        }
      });
    }

    // ⭐ تجميع المنصات - عمود المنصة [3] أو المودريتور [20] fallback
    var rawPlt   = ordersData[j][3] || ordersData[j][20] || "";
    var pltName  = detectPlatformName(rawPlt);
    monthPlatformsMap[pltName]++;
  }

  // ترتيب المنتجات تنازلياً وأخذ أفضل 10
  var sortedProducts = Object.keys(monthTopProductsMap).map(function(k) {
    return { name: k, qty: monthTopProductsMap[k] };
  }).sort(function(a, b) { return b.qty - a.qty; }).slice(0, 10);

  // ترتيب المناطق تنازلياً
  var sortedZones = Object.keys(monthZonesMap).map(function(k) {
    return { name: k, count: monthZonesMap[k].count, totalShipping: monthZonesMap[k].totalShipping };
  }).sort(function(a, b) { return b.count - a.count; });

  return {
    monthTopProducts: sortedProducts,
    monthPlatforms:   monthPlatformsMap,
    monthZonesStats:  sortedZones, // ⭐ الداتا الجديدة للداشبورد
    monthSales:       monthSales,
    monthOrderCount:  monthOrderCount,
    completedMonthCount: completedMonthCount,
    returnedCount:    returnedCount,
    moderatorStats:   moderatorStats,
    driverStats:      driverStats
  };
}

// ⭐ دالة مساعدة للبحث عن شيت الاستلامات بغض النظر عن الإيموجي أو المسافات
function getExpirySheet(ss) {
  var sheets = ss.getSheets();
  // ابحث أولاً عن شيت يحتوي على كلمة "استلام" 
  for (var i = 0; i < sheets.length; i++) {
    var sName = sheets[i].getName();
    var cleanName = sName.replace(/أ/g, "ا").replace(/إ/g, "ا").replace(/آ/g, "ا").replace(/\s/g, "");
    if (cleanName.indexOf("استلام") !== -1 && cleanName.indexOf("ارشيف") === -1) {
      return sheets[i];
    }
  }
  // إذا لم يجد، ابحث عن كلمة "صلاحيات" كبديل
  for (var i = 0; i < sheets.length; i++) {
    var sName = sheets[i].getName();
    var cleanName = sName.replace(/أ/g, "ا").replace(/إ/g, "ا").replace(/آ/g, "ا").replace(/\s/g, "");
    if (cleanName.indexOf("صلاحيات") !== -1 && cleanName.indexOf("ارشيف") === -1) {
      return sheets[i];
    }
  }
  return null;
}

    var attSheet = ss.getSheetByName("📅 سجل الحضور");
    var attendanceList = [];
    var filterEmployee = e.parameter.employee || "";
    var filterMonth = e.parameter.month || "";
    var filterDate = e.parameter.exactDate || "";
    
    if (attSheet && attSheet.getLastRow() > 1) {
      var data = attSheet.getDataRange().getValues();
      for (var i = 1; i < data.length; i++) {
        var empName = (data[i][0] || "").toString().trim();
        var dateVal = "";
        try { dateVal = Utilities.formatDate(new Date(data[i][1]), "Africa/Cairo", "yyyy-MM-dd"); } catch(ex) { dateVal = data[i][1].toString().replace("'",""); }
        
        // Apply filters
        if (filterEmployee && empName !== filterEmployee) continue;
        if (filterDate) {
          if (dateVal !== filterDate) continue;
        } else if (filterMonth && dateVal.substring(0, 7) !== filterMonth) {
          continue;
        }
        
        attendanceList.push({
          employee: empName,
          date: dateVal,
          checkIn: (data[i][2] || "").toString(),
          checkOut: (data[i][3] || "").toString(),
          hours: (data[i][4] || "").toString(),
          status: (data[i][5] || "").toString(),
          requestStatus: (data[i][6] || "").toString(),
          notes: (data[i][7] || "").toString()
        });
      }
    }
    
    // Calculate leave balance for the requested employee/month
    var leaveBalance = { paidUsed: 0, unpaidUsed: 0, paidMax: 4 };
    if (filterEmployee && filterMonth) {
      attendanceList.forEach(function(rec) {
        if (rec.date.substring(0, 7) === filterMonth && rec.employee === filterEmployee) {
          if (rec.status === "إجازة مدفوعة" && rec.requestStatus === "✅ تمت الموافقة") leaveBalance.paidUsed++;
          if (rec.status === "إجازة بدون مرتب" && rec.requestStatus === "✅ تمت الموافقة") leaveBalance.unpaidUsed++;
        }

      if (isNaN(remainingCash)) remainingCash = finalTotal;
      var driver = data[j][11];

      // أوردرات غير مصفاة للمناديب
      if (status === "تم التوصيل" && driver) {
        response.uncollectedOrders.push({
          id: data[j][0], driver: driver, total: finalTotal, remaining: remainingCash,
          payment: data[j][10], shipping: data[j][16], phone2: data[j][8] ? data[j][8].toString().replace(/'/g, "") : ""
        });
      }

      // ⭐ أوردرات في الشحن
      if (status === "في الشحن" && driver) {
        response.shippedOrders.push({
          id: data[j][0], driver: driver, total: finalTotal, remaining: remainingCash,
          payment: data[j][10], shipping: data[j][16], name: data[j][4], phone: data[j][7].toString().replace(/'/g, ""), phone2: data[j][8] ? data[j][8].toString().replace(/'/g, "") : "", address: data[j][6]
        });
      }

      var formattedResDate = data[j][12] instanceof Date
        ? Utilities.formatDate(data[j][12], "Africa/Cairo", "yyyy-MM-dd")
        : (data[j][12] ? data[j][12].toString().replace(/'/g, "").trim() : "");

      // ⭐ السجل: يعرض الطلبات حسب filterDate (اليوم المختار في السجل)
      if (rowDate === filterDate) {
        response.history.push({
          id: data[j][0], date: rowDate, time: data[j][2], name: data[j][4],
          phone: data[j][7].toString().replace(/'/g, ""), phone2: data[j][8] ? data[j][8].toString().replace(/'/g, "") : "", address: data[j][6],
          products: data[j][13], subtotal: data[j][14], discount: data[j][15],
          shipping: data[j][16], total: finalTotal, status: status,
          payment: data[j][10], seller: data[j][20], orderType: data[j][9],
          deposit: data[j][21] || 0, remaining: remainingCash,
          gov: data[j][5], platform: data[j][3], driver: driver,
          reservationDate: formattedResDate
        });
      }

      // ⭐ الأوردرات المعلقة (قيد التجهيز) - كلها بدون فلتر تاريخ
      if (status === "قيد التجهيز") {
        response.pendingOrders.push({
          id: data[j][0], name: data[j][4],
          phone: data[j][7].toString().replace(/'/g, ""), phone2: data[j][8] ? data[j][8].toString().replace(/'/g, "") : "",
          total: finalTotal, remaining: remainingCash,
          payment: data[j][10], date: rowDate,
          orderType: data[j][9],
          reservationDate: formattedResDate,
          gov: data[j][5], address: data[j][6],
          moderator: data[j][20] || "" // NEW
        });
      }

      // تجميع المنتجات الفريدة للـ autocomplete
      if (data[j][13]) {
        var pParts = data[j][13].toString().split('\n');
        pParts.forEach(function(p) { 
          var m = p.match(/^(.+?)\s*-/); 
          if (m) { uniqueProducts[m[1].trim()] = (uniqueProducts[m[1].trim()] || 0) + 1; }
        });
      }
    }

    response.products = Object.keys(uniqueProducts).map(function(n) { return { name: n }; });
  }

  return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
}

function buildDailyStats(ordersData, targetDate) {
  var todayOrders  = 0;
  var todaySales   = 0;
  var completedToday = 0;
  var todayDriverStats = {}; // ⭐ إحصائيات المناديب

  for (var j = 1; j < ordersData.length; j++) {
    var rowDate = ordersData[j][1] instanceof Date
      ? Utilities.formatDate(ordersData[j][1], "Africa/Cairo", "yyyy-MM-dd")
      : (ordersData[j][1] ? ordersData[j][1].toString().replace(/'/g, "").trim() : "");

    if (rowDate !== targetDate) continue;

    var status     = ordersData[j][18];
    var finalTotal = parseFloat(ordersData[j][17]) || 0;
    var driver     = (ordersData[j][11] || "").toString().trim();
    var shipping   = parseFloat(ordersData[j][16]) || 0;

    if (status !== "مرتجع") {
      todayOrders++;
      todaySales += finalTotal;
      
      // ⭐ حساب أرباح المناديب لليوم الحالي
      if (driver !== "") {
         if (!todayDriverStats[driver]) {
             todayDriverStats[driver] = { count: 0, profit: 0 };
         }
         todayDriverStats[driver].count++;
         todayDriverStats[driver].profit += shipping;
      }
    }
    if (status === "تم التوصيل" || status === "تم التوصيل ومُحاسب") {
      completedToday++;
    }
  }

  return { todayOrders: todayOrders, todaySales: todaySales, completedToday: completedToday, todayDriverStats: todayDriverStats };
}

        var loggedHours = parseFloat(hrStr);
        if (isNaN(loggedHours)) {
          var hMatch = hrStr.match(/(\d+(?:\.\d+)?)\s*ساعة/);
          if (hMatch) loggedHours = parseFloat(hMatch[1]);
        }
        
        if (isNaN(loggedHours) || loggedHours <= 0) continue;
        
        var bestIn = inStr;
        var bestOut = outStr;
        var foundMatch = false;
        
        var ampm = ["AM", "PM"];
        for (var a1 = 0; a1 < 2; a1++) {
          for (var a2 = 0; a2 < 2; a2++) {
            var testIn = inStr.replace(/AM|PM/gi, "").trim() + " " + ampm[a1];
            var testOut = outStr.replace(/AM|PM/gi, "").trim() + " " + ampm[a2];
            
            try {
              var d1 = new Date("2026-01-01 " + testIn);
              var d2 = new Date("2026-01-01 " + testOut);
              var diffMs = d2 - d1;
              if (diffMs < 0) diffMs += 24 * 60 * 60 * 1000;
              var calcH = diffMs / (1000 * 60 * 60);
              
              if (Math.abs(calcH - loggedHours) <= 0.1) {
                bestIn = testIn;
                bestOut = testOut;
                foundMatch = true;
                break;
              }
            } catch(e) {}
          }
          if (foundMatch) break;
        }
        
        if (foundMatch && (bestIn !== inStr || bestOut !== outStr)) {
          attSheet.getRange(a + 1, 3).setValue(bestIn);
          attSheet.getRange(a + 1, 4).setValue(bestOut);
          fixedCount++;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({success: true, fixedCount: fixedCount})).setMimeType(ContentService.MimeType.JSON);
    }


  } finally { lock.releaseLock(); }
}

function updateCustomerDB(sheet, name, phone, gov, address, amount, date) {
  var data       = sheet.getDataRange().getValues();
  var cleanPhone = phone.toString().replace(/'/g, "").trim();
  for (var i = 1; i < data.length; i++) {
    if (data[i][1].toString().replace(/'/g, "").trim() === cleanPhone) {
      var newCount = (parseInt(data[i][4]) || 0) + 1;
      var newTotal = (parseFloat(data[i][5]) || 0) + parseFloat(amount);
      sheet.getRange(i+1, 5, 1, 3).setValues([[newCount, newTotal, date]]);
      return;
    }
  }
  sheet.appendRow([name, "'" + cleanPhone, gov, address, 1, amount, date]);
}

function subtractFromCustomerDB(sheet, phone, amount) {
  if (!sheet) return;
  var data       = sheet.getDataRange().getValues();
  var cleanPhone = phone.toString().replace(/'/g, "").trim();
  for (var i = 1; i < data.length; i++) {
    if (data[i][1].toString().replace(/'/g, "").trim() === cleanPhone) {
      var cCount = parseInt(data[i][4]) || 1;
      var cTotal = parseFloat(data[i][5]) || 0;
      sheet.getRange(i+1, 5, 1, 2).setValues([[Math.max(0, cCount - 1), Math.max(0, cTotal - amount)]]);
      return;
    }
  }
}

function calculateFinancials() {
  var ss          = SpreadsheetApp.getActiveSpreadsheet();
  var ordersSheet = ss.getSheetByName("📝 الطلبات");
  var finSheet    = ss.getSheetByName("📊 الحسابات والمالية");
  
  if (!ordersSheet || !finSheet) return;

  var data     = ordersSheet.getDataRange().getValues();
  var accounts = {};

  for (var i = 1; i < data.length; i++) {
    var status    = data[i][18];
    var driver    = data[i][11];
    var payMethod = data[i][10] ? data[i][10].toString() : "";
    var shippingCost = parseFloat(data[i][16]) || 0;
    var finalTotal   = parseFloat(data[i][17]) || 0;
    
    var remainingCash = parseFloat(data[i][22]);
    if (isNaN(remainingCash)) remainingCash = finalTotal; 

    var isCash    = payMethod.indexOf("كاش 💵") !== -1 || (payMethod.includes("كاش") && !payMethod.includes("فودافون"));
    var isDigital = payMethod.includes("إنستا") || payMethod.includes("انستاباي") || payMethod.includes("فودافون") || payMethod.includes("تحويل") || payMethod.includes("محفظة");

    if ((status === "تم التوصيل" || status === "في الشحن") && driver) {
      if (!accounts[driver]) {
        accounts[driver] = { ordersCount: 0, inTransit: 0, cashCollected: 0, shippingFees: 0, netDue: 0 };
      }
      
      if (status === "في الشحن") {
         if (isCash) { accounts[driver].inTransit += remainingCash; }
      } 
      else if (status === "تم التوصيل") {
          accounts[driver].ordersCount++;
          if (isCash) {
            accounts[driver].cashCollected += remainingCash;
            accounts[driver].shippingFees  += shippingCost;
            accounts[driver].netDue        += (remainingCash - shippingCost); 
          } else if (isDigital) {
            accounts[driver].shippingFees += shippingCost;
            accounts[driver].netDue       -= shippingCost; 
          }
      }
    }
  }

  var lastRow = finSheet.getLastRow();
  if (lastRow > 1) {
    finSheet.getRange(2, 1, lastRow - 1, 7).clearContent();
  }

  var row = 2;
  for (var drv in accounts) {
    var acc        = accounts[drv];
    var statusText = acc.netDue > 0 ? "عايزين منه" : (acc.netDue < 0 ? "ليه عندنا" : "خالص");
    
    finSheet.getRange(row, 1, 1, 7).setValues([[
      drv, acc.ordersCount, acc.inTransit, acc.cashCollected, acc.shippingFees, acc.netDue, statusText 
    ]]);
    row++;
  }
}

// تحليل كبار العملاء VIP
// ============================================================
function analyzeVIPs() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var ordersSheet = ss.getSheetByName("📝 الطلبات");
  if (!ordersSheet) return [];
  
  var data = ordersSheet.getDataRange().getValues();
  var customersMap = {};
  
  for (var i = 1; i < data.length; i++) {
    var status = data[i][18];
    if (status === "تم التوصيل" || status === "تم التوصيل ومُحاسب") {
      var phone = data[i][7].toString().replace(/'/g, "").trim();
      var name = data[i][4];
      var total = parseFloat(data[i][17]) || 0;
      
      if (!customersMap[phone]) {
        customersMap[phone] = { name: name, phone: phone, orderCount: 0, totalPaid: 0 };
      }
      customersMap[phone].orderCount++;
      customersMap[phone].totalPaid += total;
    }
  }
  
  var vipList = Object.keys(customersMap).map(function(key) {
    return customersMap[key];
  }).sort(function(a, b) {
    return b.orderCount - a.orderCount;
  });
  
  return vipList;
}

// ⭐ V18: نظام المزامنة الذكية مع Firebase (الاستلامات والمرتجعات)
// ============================================================

function syncFirebaseInventory(ss) {
  var expirySheet = getExpirySheet(ss);
  if (!expirySheet) return { success: false, error: "شيت الاستلامات والصلاحيات غير موجود" };
  
  var expData = expirySheet.getDataRange().getValues();
  if (expData.length < 2) return { success: false, error: "لا توجد بيانات كافية في شيت الصلاحيات" };
  
  var headers = expData[0];
  var findIdx = function(aliases) {
    for (var j = 0; j < aliases.length; j++) {
      var idx = headers.indexOf(aliases[j]);
      if (idx !== -1) return idx;
    }
    return -1;
  };
  
  var idxName = findIdx(["اسم المنتج", "اسم الصنف", "الاسم"]);
  var idxQty = findIdx(["الكمية"]);
  var idxExp = findIdx(["تاريخ الانتهاء", "تاريخ الصلاحية"]);
  var idxBarcode = findIdx(["الباركود"]);
  
  if (idxQty === -1 || idxBarcode === -1) return { success: false, error: "الأعمدة الأساسية (الكمية، الباركود) غير موجودة" };
  
  // 1. Fetch Firebase Data
  var fbUrl = "https://candyclubsync-default-rtdb.firebaseio.com/products.json";
  var fbResponse;
  try {
    fbResponse = UrlFetchApp.fetch(fbUrl, { muteHttpExceptions: true });
    if (fbResponse.getResponseCode() !== 200) {
      return { success: false, error: "فشل الاتصال بقاعدة بيانات الفايربيز" };
    }
  } catch (e) {
    return { success: false, error: e.toString() };
  }
  
  var fbDataRaw = JSON.parse(fbResponse.getContentText());
  var fbMap = {}; // Barcode -> total stock
  var items = Array.isArray(fbDataRaw) ? fbDataRaw : Object.values(fbDataRaw);
  
  for (var i = 0; i < items.length; i++) {
    var item = items[i];
    if (item && item.Barcode) {
      var bcode = String(item.Barcode).trim();
      var stock = parseFloat(item.Stock) || 0;
      fbMap[bcode] = stock;
    }
  }
  
  // 2. Aggregate Google Sheet Data
  var gsMap = {}; // Barcode -> { totalQty: sum, rows: [] }
  for (var i = 1; i < expData.length; i++) {
    var row = expData[i];
    var bcode = String(row[idxBarcode] || "").trim();
    if (!bcode) continue; // Skip items without barcode
    
    var qty = parseFloat(row[idxQty]) || 0;
    if (!gsMap[bcode]) {
      gsMap[bcode] = { totalQty: 0, rows: [] };
    }
    
    var expDateStr = idxExp !== -1 ? row[idxExp] : "";
    var expDateObj = new Date(expDateStr);
    if (isNaN(expDateObj.getTime())) expDateObj = new Date(8640000000000000); 
    
    gsMap[bcode].totalQty += qty;
    gsMap[bcode].rows.push({ rowIndex: i, date: expDateObj, qty: qty, name: row[idxName] });
  }
  
  // 3. Compare and Adjust
  var changesCount = 0;
  var discrepancies = [];
  
  for (var bcode in gsMap) {
    var gsTotal = gsMap[bcode].totalQty;
    var fbTotal = fbMap.hasOwnProperty(bcode) ? fbMap[bcode] : null;
    
    if (fbTotal === null) continue; // Not in Firebase, maybe deleted or new
    
    var diff = gsTotal - fbTotal;
    
    if (diff > 0) {
      // Sold! We need to deduct 'diff' from GS
      var qtyToDeduct = diff;
      var rows = gsMap[bcode].rows;
      // FIFO: sort by date ascending
      rows.sort(function(a, b) { return a.date.getTime() - b.date.getTime(); });
      
      for (var m = 0; m < rows.length; m++) {
        if (qtyToDeduct <= 0) break;
        var r = rows[m];
        if (r.qty > 0) {
          if (r.qty >= qtyToDeduct) {
            expData[r.rowIndex][idxQty] = r.qty - qtyToDeduct;
            qtyToDeduct = 0;
          } else {
            qtyToDeduct -= r.qty;
            expData[r.rowIndex][idxQty] = 0;
          }
          changesCount++;
        }
      }
      if (qtyToDeduct > 0) {
        discrepancies.push("المنتج " + rows[0].name + " ناقص في الشيت بمقدار " + qtyToDeduct);
      }
    } else if (diff < 0) {
      // Returned or added to FB! We need to add Math.abs(diff) to GS
      var qtyToAdd = Math.abs(diff);
      var rows = gsMap[bcode].rows;
      
      if (rows.length > 0) {
        // Find nearest expiry date (sort ascending) and add it there
        rows.sort(function(a, b) { return a.date.getTime() - b.date.getTime(); });
        var targetRowIndex = rows[0].rowIndex;
        var currentQty = parseFloat(expData[targetRowIndex][idxQty]) || 0;
        expData[targetRowIndex][idxQty] = currentQty + qtyToAdd;
        changesCount++;
      }
    }
  }
  
  // 4. Archive empty rows and apply changes
  if (changesCount > 0) {
    var newExpData = [expData[0]];
    for (var i = 1; i < expData.length; i++) {
      var qStr = expData[i][idxQty];
      var q = parseFloat(qStr);
      if (qStr === "" || isNaN(q) || q > 0) {
        newExpData.push(expData[i]);
      }
    }
    
    expirySheet.clearContents();
    if (newExpData.length > 0) {
      expirySheet.getRange(1, 1, newExpData.length, newExpData[0].length).setValues(newExpData);
    }
    
    return { success: true, message: "تم تحديث " + changesCount + " منتج بنجاح" };
  } else {
    return { success: true, message: "البيانات متطابقة، لا توجد مبيعات أو مرتجعات جديدة لتسويتها" };
  }
}

// ============================================================
function setupAutomaticSync() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var triggers = ScriptApp.getProjectTriggers();
  var exists = false;
  
  for (var i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === 'autoSyncTriggerHandler') {
      exists = true;
      break;
    }
  }
  
  if (!exists) {
    // تفعيل المزامنة كل ساعة واحدة
    ScriptApp.newTrigger('autoSyncTriggerHandler')
      .timeBased()
      .everyHours(1)
      .create();
    SpreadsheetApp.getUi().alert("تم تفعيل المزامنة الأوتوماتيكية بنجاح! السكريبت هيشتغل لوحده كل ساعة يقارن الفايربيز ويظبط شيت الصلاحيات.");
  } else {
    SpreadsheetApp.getUi().alert("المزامنة الأوتوماتيكية مفعلة بالفعل مسبقاً.");
  }
}

function runSyncNow() {

  }
}

function autoSyncTriggerHandler() {
  var lock = LockService.getScriptLock();

  if (lock.tryLock(30000)) {
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var result = syncFirebaseInventory(ss);
      if (result.success && result.message && result.message.indexOf("تحديث 0") === -1 && result.message.indexOf("متطابقة") === -1) {
        // يمكن حفظ سجل بنجاح المزامنة هنا إذا أردت
      }
    } catch(e) {
      // تم إزالة سجل التنبيهات
    } finally {
      lock.releaseLock();
    }
  }
}
