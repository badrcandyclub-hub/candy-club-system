// ⭐ إضافة دالة التفاعل اليدوي: عشان لو غيرت الحالة من الإكسيل بإيدك، اللون يتغير والمالية تتحدث فوراً
function onEdit(e) {
  if (!e || !e.range) return;
  var sheet = e.range.getSheet();
  if (sheet.getName() !== "📝 الطلبات") return;
  
  var row = e.range.getRow();
  var col = e.range.getColumn();
  
  // العمود 19 هو "حالة الطلب" (S)
  if (col === 19 && row > 1) {
    var status = e.value;
    var oldStatus = e.oldValue;
    var rowRange = sheet.getRange(row, 1, 1, sheet.getLastColumn());
    
    if (status === "مرتجع") {
        rowRange.setBackground("#FEE2E2"); // أحمر فاتح
        if (oldStatus !== "مرتجع") {
            var phoneToSub = sheet.getRange(row, 8).getValue().toString().replace(/'/g, "").trim();
            var amtToSub = parseFloat(sheet.getRange(row, 18).getValue()) || 0;
            subtractFromCustomerDB(SpreadsheetApp.getActiveSpreadsheet().getSheetByName("👥 قاعدة العملاء"), phoneToSub, amtToSub);
        }
    } else if (status === "تم التوصيل ومُحاسب") {
        rowRange.setBackground("#E8F5E9"); // أخضر فاتح مريح
    } else if (status === "تم التوصيل") {
        rowRange.setBackground("#D4EDDA"); // أخضر باهت (قيد التحصيل)
    } else if (status === "في الشحن" || status === "قيد التجهيز") {
        rowRange.setBackground("#FFFFFF"); // أبيض طبيعي
    }
    
    // تحديث المالية في الخلفية
    // ============================================================
  // 11. شيت الحضور والانصراف (نظام HR)
  // ============================================================
  var attendanceSheet = ss.getSheetByName("📅 سجل الحضور");
  if (!attendanceSheet) {
      attendanceSheet = ss.insertSheet("📅 سجل الحضور");
      var attHeaders = ["اسم الموظف", "التاريخ", "وقت الحضور", "وقت الانصراف", "إجمالي الساعات", "الحالة", "ملاحظات"];
      attendanceSheet.getRange(1, 1, 1, attHeaders.length).setValues([attHeaders])
           .setBackground("#00897b").setFontColor("white").setFontFamily(headerFont).setFontSize(14).setFontWeight("bold")
           .setHorizontalAlignment("center").setVerticalAlignment("middle").setWrap(true);
      attendanceSheet.setRowHeight(1, 45);
      attendanceSheet.setFrozenRows(1);
      attendanceSheet.setColumnWidth(1, 180);
      attendanceSheet.setColumnWidth(2, 140);
      attendanceSheet.setColumnWidth(3, 120);
      attendanceSheet.setColumnWidth(4, 120);
      attendanceSheet.setColumnWidth(5, 120);
      attendanceSheet.setColumnWidth(6, 140);
      attendanceSheet.setColumnWidth(7, 250);
  }

  calculateFinancials();
  }
}

// ⭐ V15.1: دالة مساعدة للتعرف على المنصة من النص

function detectPlatformName(rawPlatform) {
  var plt = (rawPlatform || "").toString().toLowerCase();
  if (plt.includes('واتس') || plt.includes('whats')) return 'واتساب';
  if (plt.includes('انستا') || plt.includes('insta') || plt.includes('انستجرام') || plt.includes('إنستجرام')) return 'انستجرام';
  if (plt.includes('فيس') || plt.includes('face') || plt.includes('facebook')) return 'فيسبوك';
  if (plt.includes('تيك') || plt.includes('tiktok')) return 'تيك توك';
  return 'أخرى';
}

// ⭐ V15.1: دالة مستقلة لحساب تقرير شهر محدد
// targetMonth = "2026-06" مثلاً
// ============================================================
function buildMonthlyReport(ordersData, targetMonth) {
  var monthTopProductsMap = {};
  var monthPlatformsMap   = { 'واتساب': 0, 'انستجرام': 0, 'فيسبوك': 0, 'تيك توك': 0, 'أخرى': 0 };
  var monthZonesMap       = {}; // ⭐ تحليل مناطق الشحن
  var monthSales          = 0;
  var completedMonthCount = 0;
  var monthOrderCount     = 0;
  var returnedCount       = 0;
  var moderatorStats      = {}; // ⭐ حساب المودريتور
  var driverStats         = {}; // ⭐ حساب المناديب

  for (var j = 1; j < ordersData.length; j++) {
    var rowDate = ordersData[j][1] instanceof Date
      ? Utilities.formatDate(ordersData[j][1], "Africa/Cairo", "yyyy-MM-dd")
      : (ordersData[j][1] ? ordersData[j][1].toString().replace(/'/g, "").trim() : "");

    var status     = ordersData[j][18];
    var finalTotal = parseFloat(ordersData[j][17]) || 0;
    var shipping   = parseFloat(ordersData[j][16]) || 0;
    var mod        = (ordersData[j][20] || "").toString().trim(); // عمود 21 هو المودريتور
    var driver     = (ordersData[j][11] || "").toString().trim(); // عمود 12 هو المندوب

    if (status !== "مرتجع" && mod !== "") {
      if (!moderatorStats[mod]) {
          moderatorStats[mod] = { monthSales: 0, monthOrderCount: 0, totalSales: 0, totalCount: 0 };
      }
      moderatorStats[mod].totalSales += finalTotal;
      moderatorStats[mod].totalCount++;
      
      if (rowDate.startsWith(targetMonth)) {
         moderatorStats[mod].monthSales += finalTotal;
         moderatorStats[mod].monthOrderCount++;
      }
    }

    // ⭐ حساب المناديب
    if (driver !== "" && (status === "تم التوصيل" || status === "تم التوصيل ومُحاسب")) {
      if (!driverStats[driver]) {
          driverStats[driver] = { monthProfit: 0, monthOrderCount: 0, totalProfit: 0, totalCount: 0 };
      }
      driverStats[driver].totalProfit += shipping;
      driverStats[driver].totalCount++;
      
      if (rowDate.startsWith(targetMonth)) {
         driverStats[driver].monthProfit += shipping;
         driverStats[driver].monthOrderCount++;
      }
    }

    // فقط سجلات الشهر المطلوب
    if (!rowDate.startsWith(targetMonth)) continue;

    monthOrderCount++;

    if (status === "مرتجع") {
      returnedCount++;
      continue; // المرتجعات لا تدخل في المبيعات
    }

    monthSales += finalTotal;

    if (status === "تم التوصيل" || status === "تم التوصيل ومُحاسب") {
      completedMonthCount++;
    }

    // ⭐ تحليل مناطق الشحن للشهر
    var zone = (ordersData[j][5] || "").toString().trim(); // عمود 6 (المنطقة/المحافظة)
    var orderType = (ordersData[j][9] || "").toString().trim();
    
    if (orderType === "استلام من الفرع" || orderType === "استلام من الفرع ") {
        zone = "استلام من الفرع";
    } else {
        if (!zone && ordersData[j][6]) {
           zone = ordersData[j][6].toString().split(/[-،,\n]/)[0].trim();
        }
        if (!zone) zone = "غير محددة";
    }

    if (!monthZonesMap[zone]) {
       monthZonesMap[zone] = { count: 0, totalShipping: 0 };
    }
    monthZonesMap[zone].count++;
    if (status !== "مرتجع") { // إضافة مصاريف الشحن فقط لو الأوردر متلغاش
        monthZonesMap[zone].totalShipping += shipping;
    }

    // ⭐ تجميع المنتجات
    if (ordersData[j][13]) {
      var parts = ordersData[j][13].toString().split('\n');
      parts.forEach(function(p) {
        var m = p.match(/^(.+?)\s*-\s*الكمية:\s*(\d+)/);
        if (m) {
          var pName = m[1].trim();
          var pQty  = parseInt(m[2]) || 1;
          monthTopProductsMap[pName] = (monthTopProductsMap[pName] || 0) + pQty;
        }
      });
    }

    // ⭐ تجميع المنصات - عمود المنصة [3] أو المودريتور [20] fallback
    var rawPlt   = ordersData[j][3] || ordersData[j][20] || "";
    var pltName  = detectPlatformName(rawPlt);
    monthPlatformsMap[pltName]++;
  }

  // ترتيب المنتجات تنازلياً وأخذ أفضل 10
  var sortedProducts = Object.keys(monthTopProductsMap).map(function(k) {
    return { name: k, qty: monthTopProductsMap[k] };
  }).sort(function(a, b) { return b.qty - a.qty; }).slice(0, 10);

  // ترتيب المناطق تنازلياً
  var sortedZones = Object.keys(monthZonesMap).map(function(k) {
    return { name: k, count: monthZonesMap[k].count, totalShipping: monthZonesMap[k].totalShipping };
  }).sort(function(a, b) { return b.count - a.count; });

  return {
    monthTopProducts: sortedProducts,
    monthPlatforms:   monthPlatformsMap,
    monthZonesStats:  sortedZones, // ⭐ الداتا الجديدة للداشبورد
    monthSales:       monthSales,
    monthOrderCount:  monthOrderCount,
    completedMonthCount: completedMonthCount,
    returnedCount:    returnedCount,
    moderatorStats:   moderatorStats,
    driverStats:      driverStats
  };
}

// ⭐ دالة مساعدة للبحث عن شيت الاستلامات بغض النظر عن الإيموجي أو المسافات
function getExpirySheet(ss) {
  var sheets = ss.getSheets();
  for (var i = 0; i < sheets.length; i++) {
    var sName = sheets[i].getName();
    var cleanName = sName.replace(/أ/g, "ا").replace(/إ/g, "ا").replace(/آ/g, "ا").replace(/\s/g, "");
    if (cleanName.indexOf("استلام") !== -1 || cleanName.indexOf("صلاحيات") !== -1) {
      return sheets[i];
    }
  }
  return null;
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var action = e.parameter.action;

  // ⭐ V16: تسجيل الدخول
  if (action === "login") {
    var result = authenticateUser(e.parameter.username || "", e.parameter.password || "");
    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  }

  // ============================================================
  // ⭐ HR: جلب بيانات الحضور والانصراف
  // ============================================================
  if (action === "getAttendance") {
    var attSheet = ss.getSheetByName("📅 سجل الحضور");
    var attendanceList = [];
    var filterEmployee = e.parameter.employee || "";
    var filterMonth = e.parameter.month || "";
    var filterDate = e.parameter.exactDate || "";
    
    if (attSheet && attSheet.getLastRow() > 1) {
      var data = attSheet.getDataRange().getValues();
      for (var i = 1; i < data.length; i++) {
        var empName = (data[i][0] || "").toString().trim();
        var dateVal = "";
        try { dateVal = Utilities.formatDate(new Date(data[i][1]), "Africa/Cairo", "yyyy-MM-dd"); } catch(ex) { dateVal = data[i][1].toString().replace("'",""); }
        
        // Apply filters
        if (filterEmployee && empName !== filterEmployee) continue;
        if (filterDate) {
          if (dateVal !== filterDate) continue;
        } else if (filterMonth && dateVal.substring(0, 7) !== filterMonth) {
          continue;
        }
        
        attendanceList.push({
          employee: empName,
          date: dateVal,
          checkIn: (data[i][2] || "").toString(),
          checkOut: (data[i][3] || "").toString(),
          hours: (data[i][4] || "").toString(),
          status: (data[i][5] || "").toString(),
          notes: (data[i][6] || "").toString()
        });
      }
    }
    
    // Calculate leave balance for the requested employee/month
    var leaveBalance = { paidUsed: 0, unpaidUsed: 0, paidMax: 4 };
    if (filterEmployee && filterMonth) {
      attendanceList.forEach(function(rec) {
        if (rec.date.substring(0, 7) === filterMonth && rec.employee === filterEmployee) {
          if (rec.status === "إجازة مدفوعة" && rec.notes.indexOf("مرفوضة") === -1) leaveBalance.paidUsed++;
          if (rec.status === "إجازة بدون مرتب" && rec.notes.indexOf("مرفوضة") === -1) leaveBalance.unpaidUsed++;
        }
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({ attendance: attendanceList, leaveBalance: leaveBalance })).setMimeType(ContentService.MimeType.JSON);
  }

  // ⭐ V16: جلب قائمة المستخدمين (للمدير)
  if (action === "getUsers") {
    var usersSheet = ss.getSheetByName("👥 المستخدمين");
    var usersList = [];
    if (usersSheet && usersSheet.getLastRow() > 1) {
      var data = usersSheet.getDataRange().getValues();
      for (var i = 1; i < data.length; i++) {
        if (data[i][0]) {
          usersList.push({
            username: data[i][0],
            password: data[i][1],
            displayName: data[i][2],
            permissions: data[i][3],
            status: data[i][4],
            lastLogin: data[i][5]
          });
        }
      }
    }
    return ContentService.createTextOutput(JSON.stringify({ users: usersList })).setMimeType(ContentService.MimeType.JSON);
  }

  // ⭐ جلب بيانات المخازن والأذونات
  if (action === "getInventoryLogs") {
      var invSheet = ss.getSheetByName("📦 أذونات المخازن");
      var invLogs = [];
      var searchQuery = (e.parameter.q || "").toLowerCase().trim();
      if (invSheet) {
          var data = invSheet.getDataRange().getValues();
          
          if (searchQuery !== "") {
              // Search mode: find exact ID or matches
              for (var i = 1; i < data.length; i++) {
                  if (data[i][0].toString().toLowerCase().includes(searchQuery)) {
                      invLogs.push({
                          logId: data[i][0], from: data[i][1], to: data[i][2],
                          regName: data[i][3], timestamp: data[i][4], items: data[i][5], notes: data[i][6]
                      });
                  }
              }
          } else {
              // Default mode: return only the last 20 logs to save bandwidth
              var startIdx = Math.max(1, data.length - 20);
              for (var i = startIdx; i < data.length; i++) {
                  invLogs.push({
                      logId: data[i][0], from: data[i][1], to: data[i][2],
                      regName: data[i][3], timestamp: data[i][4], items: data[i][5], notes: data[i][6]
                  });
              }
          }
      }
      return ContentService.createTextOutput(JSON.stringify(invLogs)).setMimeType(ContentService.MimeType.JSON);
  }

  // ⭐ البحث الشامل
  if (action === "globalSearch") {
      var keyword = (e.parameter.query || "").toLowerCase().trim();
      var ordersSheet = ss.getSheetByName("📝 الطلبات");
      var searchResults = [];
      if (ordersSheet && keyword.length > 0) {
          var data = ordersSheet.getDataRange().getValues();
          for (var i = 1; i < data.length; i++) {
              var id    = data[i][0].toString().toLowerCase();
              var name  = data[i][4].toString().toLowerCase();
              var phone = data[i][7].toString().replace(/'/g, "");
              
              if (id.includes(keyword) || name.includes(keyword) || phone.includes(keyword)) {
                  var rawDate = data[i][1];
                  var formattedDate = rawDate instanceof Date
                    ? Utilities.formatDate(rawDate, "Africa/Cairo", "yyyy-MM-dd")
                    : (rawDate ? rawDate.toString().replace(/'/g, "").trim() : "");
                  var formattedResDate = data[i][12] instanceof Date
                    ? Utilities.formatDate(data[i][12], "Africa/Cairo", "yyyy-MM-dd")
                    : (data[i][12] ? data[i][12].toString().replace(/'/g, "").trim() : "");
                  searchResults.push({
                      id: data[i][0], date: formattedDate, time: data[i][2],
                      name: data[i][4], phone: phone, phone2: data[i][8] ? data[i][8].toString().replace(/'/g, "") : "", address: data[i][6],
                      products: data[i][13], subtotal: data[i][14], discount: data[i][15],
                      shipping: data[i][16], total: parseFloat(data[i][17]) || 0,
                      status: data[i][18], payment: data[i][10], seller: data[i][20],
                      orderType: data[i][9], deposit: data[i][21] || 0,
                      remaining: data[i][22] || (parseFloat(data[i][17]) || 0),
                      gov: data[i][5], platform: data[i][3], driver: data[i][11],
                      reservationDate: formattedResDate
                  });
              }
          }
      }
      return ContentService.createTextOutput(JSON.stringify(searchResults.reverse())).setMimeType(ContentService.MimeType.JSON);
  }

  // ⭐ V15.1: جلب بيانات العملاء منفصلاً (Lazy Load)
  if (action === "getCustomers") {
    var customerSheet = ss.getSheetByName("👥 قاعدة العملاء");
    var custResult = [];
    // VIP is based on completed deliveries, not on every order that was merely created.
    // Build the lookup once, then attach the completed-order count to each customer.
    var vipLookup = {};
    analyzeVIPs().forEach(function(customer) {
      vipLookup[customer.phone] = customer;
    });
    if (customerSheet) {

      for (var i = 1; i < custData.length; i++) {
        if (custData[i][0]) {

          custResult.push({
            name: custData[i][0], phone: custData[i][1], gov: custData[i][2],
            address: custData[i][3], count: custData[i][4], total: custData[i][5], lastDate: custData[i][6],

    }

  }

  // ⭐ V15.2: جلب بيانات الاستلامات والصلاحيات
  if (action === "getExpiries") {
    var expirySheet = getExpirySheet(ss);
    var expResult = [];
    if (expirySheet) {
      var expData = expirySheet.getDataRange().getValues();
      if (expData.length > 0) {
        var headers = expData[0];
        
        // Helper inline function to support fallback header names (aliases)
        var findIdx = function(aliases) {
          for (var j = 0; j < aliases.length; j++) {
            var idx = headers.indexOf(aliases[j]);
            if (idx !== -1) return idx;
          }
          return -1;
        };

        var idxName = findIdx(["اسم المنتج", "اسم الصنف", "الاسم"]);
        var idxQty = findIdx(["الكمية"]);
        var idxExp = findIdx(["تاريخ الانتهاء", "تاريخ الصلاحية"]);
        var idxLoc = findIdx(["المكان / المورد", "مكان التخزين", "المكان"]);
        var idxRegName = findIdx(["اسم المسجل"]);
        var idxRegDate = findIdx(["تاريخ التسجيل", "تاريخ الادخال"]);
        var idxReceiver = findIdx(["المستلم", "اسم المستلم"]);
        var idxNotes = findIdx(["ملاحظات"]);
        var idxOrig = findIdx(["السعر الأساسي", "السعر الأصلي"]);
        var idxOffer = findIdx(["سعر العرض"]);
        var idxStatus = findIdx(["الحالة"]);
        var idxBarcode = findIdx(["الباركود"]);

        for (var i = 1; i < expData.length; i++) {
          var row = expData[i];
          if (idxName !== -1 && row[idxName]) {

  var userPerms = (e.parameter.permissions || "ALL").toString();

    return permsList.indexOf(perm) !== -1;

  var ordersSheet  = (hasPerm("create") || hasPerm("history") || hasPerm("reports") || hasPerm("financials") || hasPerm("shipping") || hasPerm("customers")) ? ss.getSheetByName("📝 الطلبات") : null;

  var response = {

    catalog: [], outOfStock: [], customers: [],
    todayOrders: 0, todaySales: 0, monthSales: 0,
    completedMonthCount: 0, completedTodayCount: 0,
    history: [], pendingOrders: [], financials: [],
    monthTopProducts: [], monthPlatforms: {}, uncollectedOrders: [],
    suspendedOrders: [], shippedOrders: []
  };

  // ⭐ قراءة الطلبات المعلقة
  if (draftsSheet && draftsSheet.getLastRow() > 1) {
    var draftsData = draftsSheet.getDataRange().getValues();
    for (var i = 1; i < draftsData.length; i++) {
      if (draftsData[i][0]) {
        try {
          var parsedDraft = JSON.parse(draftsData[i][2]);
          response.suspendedOrders.push(parsedDraft);
        } catch(err) {}
      }
    }

  // ⭐ الحسابات المالية
  if (finSheet) {
    var finData = finSheet.getDataRange().getValues();
    for (var i = 1; i < finData.length; i++) {
      if (finData[i][0]) {
        response.financials.push({
          name: finData[i][0], ordersCount: finData[i][1], inTransit: finData[i][2], 
          cashCollected: finData[i][3], shippingFees: finData[i][4], netDue: finData[i][5], statusText: finData[i][6]
        });
      }

  // ⭐ العملاء - يُرسلون فقط إذا لم يكن طلب Lazy منفصل

    var custData = customerSheet.getDataRange().getValues();
    for (var i = 1; i < custData.length; i++) {
      if (custData[i][0]) {
        response.customers.push({
          name: custData[i][0], phone: custData[i][1], gov: custData[i][2],

        });
      }
    }
  }

  // ⭐ الكتالوج
  if (catalogSheet) {
    var cData = catalogSheet.getDataRange().getValues();
    for (var i = 1; i < cData.length; i++) {
      if (cData[i][0]) {

  }

  // ⭐ النواقص
  if (oosSheet) {
    var oosData = oosSheet.getDataRange().getValues();
    for (var i = 1; i < oosData.length; i++) {
      if (oosData[i][0]) {
        response.outOfStock.push({ customer: oosData[i][0], phone: oosData[i][1], product: oosData[i][2], reason: oosData[i][3], date: oosData[i][4] });
      }
    }
  }

  // ⭐ الإعدادات والشحن والمناديب والمودريتور
  if (settingsSheet) {
    var sData = settingsSheet.getDataRange().getValues();
    for (var i = 1; i < sData.length; i++) {
      if (sData[i][0])  response.alex.push({ name: sData[i][0], price: sData[i][1], type: sData[i][2], duration: sData[i][3] });
      if (sData[i][6])  response.govs.push({ name: sData[i][6], price: sData[i][7], type: sData[i][8], duration: sData[i][9] });
      if (sData[i][12]) response.couriers.push({ name: sData[i][12], phone: sData[i][13].toString() });
      if (sData[i][18]) response.moderators.push(sData[i][18].toString());
    }
  }

  // ============================================================
  // ⭐ V15.1: المنطق الرئيسي لقراءة الطلبات
  // ============================================================
  if (ordersSheet) {
    var data = ordersSheet.getDataRange().getValues();
    var uniqueProducts = {};

    // ⭐ اليوم الفعلي (للإحصائيات الحقيقية لليوم)
    var todayReal = Utilities.formatDate(new Date(), "Africa/Cairo", "yyyy-MM-dd");
    var todayMonth = todayReal.substring(0, 7);

    // ⭐ V15.1: بناء التقرير الشهري باستخدام الدالة المستقلة
    // - إذا كان طلب تقرير شهري (filterDate = YYYY-MM-01): يحسب للشهر المطلوب
    // - خلاف ذلك: يحسب للشهر الحالي الفعلي

    response.monthOrderCount     = monthlyReport.monthOrderCount;
    response.returnedCount       = monthlyReport.returnedCount;
    response.moderatorStats      = monthlyReport.moderatorStats; // NEW
    response.driverStats         = monthlyReport.driverStats;    // NEW
    response.monthZonesStats     = monthlyReport.monthZonesStats; // FIX: Return the zones data


    // ⭐ إحصائيات اليوم الحقيقي (دايماً اليوم الفعلي، مش filterDate)
    var todayReport = buildDailyStats(data, todayReal);
    response.todayOrders          = todayReport.todayOrders;
    response.todaySales           = todayReport.todaySales;
    response.completedTodayCount  = todayReport.completedToday;
    response.todayDriverStats     = todayReport.todayDriverStats; // ⭐ إضافة إحصائيات مناديب اليوم

    // ⭐ قراءة السجل (history) للـ filterDate المطلوبة
    for (var j = 1; j < data.length; j++) {
      var rowDate = data[j][1] instanceof Date
        ? Utilities.formatDate(data[j][1], "Africa/Cairo", "yyyy-MM-dd")
        : (data[j][1] ? data[j][1].toString().replace(/'/g, "").trim() : "");

      var status      = data[j][18];
      var finalTotal  = parseFloat(data[j][17]) || 0;
      var remainingCash = parseFloat(data[j][22]);
      if (isNaN(remainingCash)) remainingCash = finalTotal;
      var driver = data[j][11];

      // أوردرات غير مصفاة للمناديب
      if (status === "تم التوصيل" && driver) {
        response.uncollectedOrders.push({
          id: data[j][0], driver: driver, total: finalTotal, remaining: remainingCash,
          payment: data[j][10], shipping: data[j][16], phone2: data[j][8] ? data[j][8].toString().replace(/'/g, "") : ""
        });
      }

      // ⭐ أوردرات في الشحن
      if (status === "في الشحن" && driver) {
        response.shippedOrders.push({
          id: data[j][0], driver: driver, total: finalTotal, remaining: remainingCash,
          payment: data[j][10], shipping: data[j][16], name: data[j][4], phone: data[j][7].toString().replace(/'/g, ""), phone2: data[j][8] ? data[j][8].toString().replace(/'/g, "") : "", address: data[j][6]
        });
      }

      var formattedResDate = data[j][12] instanceof Date
        ? Utilities.formatDate(data[j][12], "Africa/Cairo", "yyyy-MM-dd")
        : (data[j][12] ? data[j][12].toString().replace(/'/g, "").trim() : "");

      // ⭐ السجل: يعرض الطلبات حسب filterDate (اليوم المختار في السجل)
      if (rowDate === filterDate) {
        response.history.push({
          id: data[j][0], date: rowDate, time: data[j][2], name: data[j][4],
          phone: data[j][7].toString().replace(/'/g, ""), phone2: data[j][8] ? data[j][8].toString().replace(/'/g, "") : "", address: data[j][6],
          products: data[j][13], subtotal: data[j][14], discount: data[j][15],
          shipping: data[j][16], total: finalTotal, status: status,
          payment: data[j][10], seller: data[j][20], orderType: data[j][9],
          deposit: data[j][21] || 0, remaining: remainingCash,
          gov: data[j][5], platform: data[j][3], driver: driver,
          reservationDate: formattedResDate
        });
      }

      // ⭐ الأوردرات المعلقة (قيد التجهيز) - كلها بدون فلتر تاريخ
      if (status === "قيد التجهيز") {
        response.pendingOrders.push({
          id: data[j][0], name: data[j][4],
          phone: data[j][7].toString().replace(/'/g, ""), phone2: data[j][8] ? data[j][8].toString().replace(/'/g, "") : "",
          total: finalTotal, remaining: remainingCash,
          payment: data[j][10], date: rowDate,
          orderType: data[j][9],
          reservationDate: formattedResDate,
          gov: data[j][5], address: data[j][6],
          moderator: data[j][20] || "" // NEW
        });
      }

      // تجميع المنتجات الفريدة للـ autocomplete
      if (data[j][13]) {
        var pParts = data[j][13].toString().split('\n');
        pParts.forEach(function(p) { 
          var m = p.match(/^(.+?)\s*-/); 
          if (m) { uniqueProducts[m[1].trim()] = (uniqueProducts[m[1].trim()] || 0) + 1; }
        });
      }
    }

    response.products = Object.keys(uniqueProducts).map(function(n) { return { name: n }; });
  }

  return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON);
}

function buildDailyStats(ordersData, targetDate) {
  var todayOrders  = 0;
  var todaySales   = 0;
  var completedToday = 0;
  var todayDriverStats = {}; // ⭐ إحصائيات المناديب

  for (var j = 1; j < ordersData.length; j++) {
    var rowDate = ordersData[j][1] instanceof Date
      ? Utilities.formatDate(ordersData[j][1], "Africa/Cairo", "yyyy-MM-dd")
      : (ordersData[j][1] ? ordersData[j][1].toString().replace(/'/g, "").trim() : "");

    if (rowDate !== targetDate) continue;

    var status     = ordersData[j][18];
    var finalTotal = parseFloat(ordersData[j][17]) || 0;
    var driver     = (ordersData[j][11] || "").toString().trim();
    var shipping   = parseFloat(ordersData[j][16]) || 0;

    if (status !== "مرتجع") {
      todayOrders++;
      todaySales += finalTotal;
      
      // ⭐ حساب أرباح المناديب لليوم الحالي
      if (driver !== "") {
         if (!todayDriverStats[driver]) {
             todayDriverStats[driver] = { count: 0, profit: 0 };
         }
         todayDriverStats[driver].count++;
         todayDriverStats[driver].profit += shipping;
      }
    }
    if (status === "تم التوصيل" || status === "تم التوصيل ومُحاسب") {
      completedToday++;
    }
  }

  return { todayOrders: todayOrders, todaySales: todaySales, completedToday: completedToday, todayDriverStats: todayDriverStats };
}

      var regName = e.parameter.regName;
      var timestamp = Utilities.formatDate(new Date(), "Africa/Cairo", "yyyy-MM-dd hh:mm:ss a");
      var items = e.parameter.items; // JSON string or plain text
      var notes = e.parameter.notes;

      invSheet.appendRow([logId, from, to, regName, timestamp, items, notes]);
      return ContentService.createTextOutput("success");
    }

    // ⭐ V16: إدارة المستخدمين
    else if (action === "addUser") {
      var usersSheet = ss.getSheetByName("👥 المستخدمين");
      if (!usersSheet) return ContentService.createTextOutput(JSON.stringify({success: false, error: "شيت المستخدمين غير موجود"})).setMimeType(ContentService.MimeType.JSON);
      // التحقق من عدم تكرار اسم المستخدم
      var uData = usersSheet.getDataRange().getValues();
      for (var u = 1; u < uData.length; u++) {
        if (uData[u][0].toString().trim() === e.parameter.username.trim()) {
          return ContentService.createTextOutput(JSON.stringify({success: false, error: "اسم المستخدم موجود بالفعل"})).setMimeType(ContentService.MimeType.JSON);
        }
      }
      usersSheet.appendRow([e.parameter.username, e.parameter.password, e.parameter.displayName, e.parameter.permissions, "نشط", ""]);
      // Ensure password column (B) is treated as text to keep leading zeros
      usersSheet.getRange(usersSheet.getLastRow(), 2).setNumberFormat("@");
      return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
    }

    else if (action === "updateUser") {
      var usersSheet = ss.getSheetByName("👥 المستخدمين");
      if (!usersSheet) return ContentService.createTextOutput(JSON.stringify({success: false, error: "شيت المستخدمين غير موجود"})).setMimeType(ContentService.MimeType.JSON);
      var uData = usersSheet.getDataRange().getValues();
      for (var u = 1; u < uData.length; u++) {
        if (uData[u][0].toString().trim() === e.parameter.username.trim()) {
          if (e.parameter.displayName) usersSheet.getRange(u + 1, 3).setValue(e.parameter.displayName);
          if (e.parameter.permissions) usersSheet.getRange(u + 1, 4).setValue(e.parameter.permissions);
          if (e.parameter.status) usersSheet.getRange(u + 1, 5).setValue(e.parameter.status);
          if (e.parameter.newPassword && e.parameter.newPassword !== "") {
            usersSheet.getRange(u + 1, 2).setValue(e.parameter.newPassword);
            // Preserve leading zeros in updated password
            usersSheet.getRange(u + 1, 2).setNumberFormat("@");
          }
          return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({success: false, error: "المستخدم غير موجود"})).setMimeType(ContentService.MimeType.JSON);
    }

    else if (action === "deleteUser") {
      var usersSheet = ss.getSheetByName("👥 المستخدمين");
      if (!usersSheet) return ContentService.createTextOutput(JSON.stringify({success: false, error: "شيت المستخدمين غير موجود"})).setMimeType(ContentService.MimeType.JSON);
      if (e.parameter.username === "admin") {
        return ContentService.createTextOutput(JSON.stringify({success: false, error: "لا يمكن حذف حساب المدير"})).setMimeType(ContentService.MimeType.JSON);
      }
      var uData = usersSheet.getDataRange().getValues();
      for (var u = 1; u < uData.length; u++) {
        if (uData[u][0].toString().trim() === e.parameter.username.trim()) {
          usersSheet.deleteRow(u + 1);
          return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({success: false, error: "المستخدم غير موجود"})).setMimeType(ContentService.MimeType.JSON);
    }


    // ============================================================
    // ⭐ HR: تسجيل حضور
    // ============================================================
    if (action === "checkIn") {
      var attSheet = ss.getSheetByName("📅 سجل الحضور");
      if (!attSheet) {
        attSheet = ss.insertSheet("📅 سجل الحضور");
        attSheet.appendRow(["اسم الموظف", "التاريخ", "وقت الحضور", "وقت الانصراف", "إجمالي الساعات", "الحالة", "ملاحظات"]);
      }
      var empName = e.parameter.employeeName || "";
      var attDate = e.parameter.date || Utilities.formatDate(new Date(), "Africa/Cairo", "yyyy-MM-dd");
      var attTime = Utilities.formatDate(new Date(), "Africa/Cairo", "hh:mm a");
      
      // Check if already checked in today
      var attData = attSheet.getDataRange().getValues();
      for (var a = 1; a < attData.length; a++) {
        var existDate = "";
        try { existDate = Utilities.formatDate(new Date(attData[a][1]), "Africa/Cairo", "yyyy-MM-dd"); } catch(ex) { existDate = attData[a][1].toString().replace("'",""); }
        if (attData[a][0].toString().trim() === empName && existDate === attDate && attData[a][5] === "حاضر") {
          return ContentService.createTextOutput(JSON.stringify({success: false, error: "أنت مسجل حضور بالفعل اليوم"})).setMimeType(ContentService.MimeType.JSON);
        }
      }
      
      attSheet.appendRow([empName, "'" + attDate, attTime, "", "", "حاضر", ""]);
      attSheet.getRange(attSheet.getLastRow(), 1, 1, 7).setBackground("#e8f5e9");
      return ContentService.createTextOutput(JSON.stringify({success: true, message: "تم تسجيل الحضور بنجاح", time: attTime})).setMimeType(ContentService.MimeType.JSON);
    }

    // ============================================================
    // ⭐ HR: تسجيل انصراف
    // ============================================================
    if (action === "checkOut") {
      var attSheet = ss.getSheetByName("📅 سجل الحضور");
      if (!attSheet) return ContentService.createTextOutput(JSON.stringify({success: false, error: "شيت الحضور غير موجود"})).setMimeType(ContentService.MimeType.JSON);
      
      var empName = e.parameter.employeeName || "";
      var attDate = e.parameter.date || Utilities.formatDate(new Date(), "Africa/Cairo", "yyyy-MM-dd");
      var outTime = Utilities.formatDate(new Date(), "Africa/Cairo", "hh:mm a");
      
      var attData = attSheet.getDataRange().getValues();
      for (var a = 1; a < attData.length; a++) {
        var existDate = "";
        try { existDate = Utilities.formatDate(new Date(attData[a][1]), "Africa/Cairo", "yyyy-MM-dd"); } catch(ex) { existDate = attData[a][1].toString().replace("'",""); }
        if (attData[a][0].toString().trim() === empName && existDate === attDate && attData[a][5] === "حاضر" && !attData[a][3]) {
          attSheet.getRange(a + 1, 4).setValue(outTime);
          
          // حساب الساعات
          var inTime = attData[a][2].toString();
          var inDate = new Date("2026-01-01 " + inTime);
          var outDate = new Date("2026-01-01 " + outTime);
          var diffMs = outDate - inDate;
          if (diffMs < 0) diffMs += 24 * 60 * 60 * 1000;
          var totalH = Math.floor(diffMs / (1000 * 60 * 60));
          var totalM = Math.floor((diffMs % (1000 * 60 * 60)) / (1000 * 60));
          var timeString = totalH + " ساعة";
          if (totalM > 0) timeString += " و " + totalM + " دقيقة";
          
          attSheet.getRange(a + 1, 5).setValue(timeString);
          return ContentService.createTextOutput(JSON.stringify({success: true, message: "تم تسجيل الانصراف بنجاح", time: outTime, hours: timeString})).setMimeType(ContentService.MimeType.JSON);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({success: false, error: "لم يتم العثور على سجل حضور مفتوح لهذا اليوم"})).setMimeType(ContentService.MimeType.JSON);
    }

    // ============================================================
    // ⭐ HR: طلب إجازة
    // ============================================================
    if (action === "requestLeave") {
      var attSheet = ss.getSheetByName("📅 سجل الحضور");
      if (!attSheet) return ContentService.createTextOutput(JSON.stringify({success: false, error: "شيت الحضور غير موجود"})).setMimeType(ContentService.MimeType.JSON);
      
      var empName = e.parameter.employeeName || "";
      var leaveDate = e.parameter.date || "";
      var leaveType = e.parameter.leaveType || "إجازة مدفوعة";
      var notes = e.parameter.notes || "";
      
      var attData = attSheet.getDataRange().getValues();
      for (var a = 1; a < attData.length; a++) {
        var existDate = "";
        try { existDate = Utilities.formatDate(new Date(attData[a][1]), "Africa/Cairo", "yyyy-MM-dd"); } catch(ex) { existDate = attData[a][1].toString().replace("'",""); }
        if (attData[a][0].toString().trim() === empName && existDate === leaveDate) {
          return ContentService.createTextOutput(JSON.stringify({success: false, error: "يوجد سجل لهذا اليوم بالفعل"})).setMimeType(ContentService.MimeType.JSON);
        }
      }
      
      var hoursCredit = leaveType === "إجازة مدفوعة" ? "8 ساعة" : "0";
      attSheet.appendRow([empName, "'" + leaveDate, "-", "-", hoursCredit, leaveType, notes + " (بانتظار الموافقة)"]);
      return ContentService.createTextOutput(JSON.stringify({success: true, message: "تم إرسال طلب الإجازة"})).setMimeType(ContentService.MimeType.JSON);
    }

    // ============================================================
    // ⭐ HR: موافقة / رفض إجازة
    // ============================================================
    if (action === "manageLeave") {
      var attSheet = ss.getSheetByName("📅 سجل الحضور");
      if (!attSheet) return ContentService.createTextOutput(JSON.stringify({success: false, error: "شيت الحضور غير موجود"})).setMimeType(ContentService.MimeType.JSON);
      
      var empName = e.parameter.employeeName || "";
      var leaveDate = e.parameter.date || "";
      var decision = e.parameter.decision || "";
      
      var attData = attSheet.getDataRange().getValues();
      for (var a = 1; a < attData.length; a++) {
        var existDate = "";
        try { existDate = Utilities.formatDate(new Date(attData[a][1]), "Africa/Cairo", "yyyy-MM-dd"); } catch(ex) { existDate = attData[a][1].toString().replace("'",""); }
        if (attData[a][0].toString().trim() === empName && existDate === leaveDate) {
          if (decision === "approve") {
            var notes = (attData[a][6] || "").toString().replace(" (بانتظار الموافقة)", "");
            attSheet.getRange(a + 1, 7).setValue(notes + " ✅ تمت الموافقة");
            attSheet.getRange(a + 1, 1, 1, 7).setBackground(attData[a][5] === "إجازة مدفوعة" ? "#e3f2fd" : "#fff3e0");
          } else {
            attSheet.getRange(a + 1, 5).setValue("0");
            attSheet.getRange(a + 1, 6).setValue("إجازة مرفوضة");
            var notes = (attData[a][6] || "").toString().replace(" (بانتظار الموافقة)", "");
            attSheet.getRange(a + 1, 7).setValue(notes + " ❌ مرفوضة");
            attSheet.getRange(a + 1, 1, 1, 7).setBackground("#ffebee");
          }
          return ContentService.createTextOutput(JSON.stringify({success: true, message: decision === "approve" ? "تمت الموافقة" : "تم الرفض"})).setMimeType(ContentService.MimeType.JSON);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({success: false, error: "لم يتم العثور على السجل"})).setMimeType(ContentService.MimeType.JSON);
    }

    // ============================================================
    // ⭐ HR: إضافة إجازة مباشرة من المدير
    // ============================================================
    if (action === "addLeaveByAdmin") {
      var attSheet = ss.getSheetByName("📅 سجل الحضور");
      if (!attSheet) return ContentService.createTextOutput(JSON.stringify({success: false, error: "شيت الحضور غير موجود"})).setMimeType(ContentService.MimeType.JSON);
      
      var empName = e.parameter.employeeName || "";
      var leaveDate = e.parameter.date || "";
      var leaveType = e.parameter.leaveType || "إجازة مدفوعة";
      var notes = e.parameter.notes || "أضيفت بواسطة المدير";
      
      var hoursCredit = leaveType === "إجازة مدفوعة" ? "8 ساعة" : "0";
      attSheet.appendRow([empName, "'" + leaveDate, "-", "-", hoursCredit, leaveType, notes + " ✅ تمت الموافقة"]);
      attSheet.getRange(attSheet.getLastRow(), 1, 1, 7).setBackground(leaveType === "إجازة مدفوعة" ? "#e3f2fd" : "#fff3e0");
      return ContentService.createTextOutput(JSON.stringify({success: true, message: "تمت إضافة الإجازة"})).setMimeType(ContentService.MimeType.JSON);
    }


  } finally { lock.releaseLock(); }
}

function updateCustomerDB(sheet, name, phone, gov, address, amount, date) {
  var data       = sheet.getDataRange().getValues();
  var cleanPhone = phone.toString().replace(/'/g, "").trim();
  for (var i = 1; i < data.length; i++) {
    if (data[i][1].toString().replace(/'/g, "").trim() === cleanPhone) {
      var newCount = (parseInt(data[i][4]) || 0) + 1;
      var newTotal = (parseFloat(data[i][5]) || 0) + parseFloat(amount);
      sheet.getRange(i+1, 5, 1, 3).setValues([[newCount, newTotal, date]]);
      return;
    }
  }
  sheet.appendRow([name, "'" + cleanPhone, gov, address, 1, amount, date]);
}

function subtractFromCustomerDB(sheet, phone, amount) {
  if (!sheet) return;
  var data       = sheet.getDataRange().getValues();
  var cleanPhone = phone.toString().replace(/'/g, "").trim();
  for (var i = 1; i < data.length; i++) {
    if (data[i][1].toString().replace(/'/g, "").trim() === cleanPhone) {
      var cCount = parseInt(data[i][4]) || 1;
      var cTotal = parseFloat(data[i][5]) || 0;
      sheet.getRange(i+1, 5, 1, 2).setValues([[Math.max(0, cCount - 1), Math.max(0, cTotal - amount)]]);
      return;
    }
  }
}

// حساب الحسابات المالية للمناديب
// ============================================================
function calculateFinancials() {
  var ss          = SpreadsheetApp.getActiveSpreadsheet();
  var ordersSheet = ss.getSheetByName("📝 الطلبات");
  var finSheet    = ss.getSheetByName("📊 الحسابات والمالية");
  
  if (!ordersSheet || !finSheet) return;

  var data     = ordersSheet.getDataRange().getValues();
  var accounts = {};

  for (var i = 1; i < data.length; i++) {
    var status    = data[i][18];
    var driver    = data[i][11];
    var payMethod = data[i][10] ? data[i][10].toString() : "";
    var shippingCost = parseFloat(data[i][16]) || 0;
    var finalTotal   = parseFloat(data[i][17]) || 0;
    
    var remainingCash = parseFloat(data[i][22]);
    if (isNaN(remainingCash)) remainingCash = finalTotal; 

    var isCash    = payMethod.indexOf("كاش 💵") !== -1 || (payMethod.includes("كاش") && !payMethod.includes("فودافون"));
    var isDigital = payMethod.includes("إنستا") || payMethod.includes("انستاباي") || payMethod.includes("فودافون") || payMethod.includes("تحويل") || payMethod.includes("محفظة");

    if ((status === "تم التوصيل" || status === "في الشحن") && driver) {
      if (!accounts[driver]) {
        accounts[driver] = { ordersCount: 0, inTransit: 0, cashCollected: 0, shippingFees: 0, netDue: 0 };
      }
      
      if (status === "في الشحن") {
         if (isCash) { accounts[driver].inTransit += remainingCash; }
      } 
      else if (status === "تم التوصيل") {
          accounts[driver].ordersCount++;
          if (isCash) {
            accounts[driver].cashCollected += remainingCash;
            accounts[driver].shippingFees  += shippingCost;
            accounts[driver].netDue        += (remainingCash - shippingCost); 
          } else if (isDigital) {
            accounts[driver].shippingFees += shippingCost;
            accounts[driver].netDue       -= shippingCost; 
          }
      }
    }
  }

  var lastRow = finSheet.getLastRow();
  if (lastRow > 1) {
    finSheet.getRange(2, 1, lastRow - 1, 7).clearContent();
  }

  var row = 2;
  for (var drv in accounts) {
    var acc        = accounts[drv];
    var statusText = acc.netDue > 0 ? "عايزين منه" : (acc.netDue < 0 ? "ليه عندنا" : "خالص");
    
    finSheet.getRange(row, 1, 1, 7).setValues([[
      drv, acc.ordersCount, acc.inTransit, acc.cashCollected, acc.shippingFees, acc.netDue, statusText 
    ]]);
    row++;
  }
}

// تحليل كبار العملاء VIP
// ============================================================
function analyzeVIPs() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var ordersSheet = ss.getSheetByName("📝 الطلبات");
  if (!ordersSheet) return [];
  
  var data = ordersSheet.getDataRange().getValues();
  var customersMap = {};
  
  for (var i = 1; i < data.length; i++) {
    var status = data[i][18];
    if (status === "تم التوصيل" || status === "تم التوصيل ومُحاسب") {
      var phone = data[i][7].toString().replace(/'/g, "").trim();
      var name = data[i][4];
      var total = parseFloat(data[i][17]) || 0;
      
      if (!customersMap[phone]) {
        customersMap[phone] = { name: name, phone: phone, orderCount: 0, totalPaid: 0 };
      }
      customersMap[phone].orderCount++;
      customersMap[phone].totalPaid += total;
    }
  }
  
  var vipList = Object.keys(customersMap).map(function(key) {
    return customersMap[key];
  }).sort(function(a, b) {
    return b.orderCount - a.orderCount;
  });
  
  return vipList;
}
